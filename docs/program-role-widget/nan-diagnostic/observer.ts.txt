(Error as any).stackTraceLimit = 100;
let sequence = 0;
let router: Router | undefined;
const routeTree = (route: any): any => route ? ({path:route.routeConfig?.path, params:route.snapshot.params, query:route.snapshot.queryParams, children:route.children.map(routeTree)}) : null;
function trace(kind:string, detail:any = {}, stack = false) {
  const body = JSON.stringify({seq:++sequence, time:Date.now(), elapsed:performance.now(), kind, url:location.href, detail, ...(stack?{stack:new Error(kind).stack}: {})}, (_key, value)=>typeof value==='number' && !Number.isFinite(value)?String(value):value);
  console.info('[NAN-DIAG] '+body);
  fetch('/__nan_trace', {method:'POST',headers:{'Content-Type':'application/json'},body,keepalive:true}).catch(()=>{});
}
trace('boot',{queueEntries:JSON.parse(localStorage.getItem('project-autosave-queue')??'[]').length});

const originalBind = ProjectFormAutosaveService.prototype.bindDraftCleanupAutosave;
ProjectFormAutosaveService.prototype.bindDraftCleanupAutosave = function(control, field, destroyRef) {
  const route = (this as any).route;
  trace('bind',{field,rootRoute:route===route.root,params:route.snapshot.params});
  const subscription = control?.valueChanges.subscribe(value=>trace('control-emission',{field,value,params:route.snapshot.params,rootRoute:route===route.root,routeTree:routeTree(route.root)},!value));
  destroyRef.onDestroy(()=>{subscription?.unsubscribe();trace('binding-destroy',{field});});
  return originalBind.call(this,control,field,destroyRef);
};

const originalOpen = XMLHttpRequest.prototype.open;
const originalSend = XMLHttpRequest.prototype.send;
const requests = new WeakMap<XMLHttpRequest,{method:string,url:string}>();
XMLHttpRequest.prototype.open = function(method:string,url:string|URL,...rest:any[]) {
  requests.set(this,{method,url:String(url)});
  return (originalOpen as any).call(this,method,url,...rest);
};
XMLHttpRequest.prototype.send = function(body) {
  const request = requests.get(this);
  if (request && /\/(projects|programs)\//.test(request.url) && request.method!=='GET') {
    trace('xhr-send',{...request,body},true);
    this.addEventListener('loadend',()=>trace('xhr-end',{...request,status:this.status,response:this.responseText?.slice(0,600)}),{once:true});
  }
  return originalSend.call(this,body);
};
